# Pythia arm only. Install AFTER a RunPod PyTorch base image, which already ships torch + CUDA.
#
# DO NOT run the repo-root setup.sh for this work. It installs sae_lens + transformer_lens for
# the gemma pipeline; nothing here imports either, and `pip install sae_lens` is the documented
# cause of torch being bumped to a wheel newer than the host driver (seen 2026-08-11: torch
# 2.13.0+cu130 onto a 12.8-driver host -> torch.cuda.is_available() False, everything silently
# on CPU). Keeping this stack small is the point.
#
# THE NAME TRAP: `pip install sparsify` installs NEURAL MAGIC's deprecated package of that name.
# EleutherAI's is published as `eai-sparsify`; the import name is still `sparsify`.
eai-sparsify==1.3.3

# torch is deliberately NOT listed -- it comes from the base image. eai-sparsify declares an
# unpinned `torch`, which any existing version satisfies, so pip will leave the image's alone.
transformers
datasets
huggingface-hub
matplotlib

# Optional but recommended:
#   wandb        -- sparsify logs FVU / dead latents / config automatically when it is present
#   bitsandbytes -- gives Adam8bit, cutting optimizer state from 4.3 GB to 1.07 GB. sparsify
#                   falls back to torch.optim.Adam with a printed notice if it is missing.
wandb
bitsandbytes
